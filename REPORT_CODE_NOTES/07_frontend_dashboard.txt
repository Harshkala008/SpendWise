const submitExpense = async (event) => {
  event.preventDefault();
  setSaving(true);
  try {
    const payload = {
      ...form,
      category: form.category === 'Other'
        ? form.categoryName.trim() || 'Other'
        : form.category
    };
    delete payload.categoryName;
    await api.post('/expenses', payload);
    setForm(makeEmptyForm());
    setNotice({ type: 'success', text: 'Expense added.' });
    await Promise.all([loadExpenses(), loadSummary()]);
  } catch (requestError) {
    setNotice({ type: 'error', text: getApiError(requestError) });
  } finally {
    setSaving(false);
  }
};

PSEUDOCODE:

prevent normal form submission
show saving state
convert custom "Other" category into a category value
remove frontend-only category field
send expense to API
reset form after success
reload list and monthly summary
show error if API call fails
hide saving state

The dashboard uses SVG components to visualize the returned summary. The
spending chart shows the monthly total and the pie chart converts category

MAIN CODE EXCERPT:

const items = breakdown.slice(0, 5);
const total = items.reduce((sum, item) => sum + item.total, 0);

const slices = items.reduce((result, item, index) => {
  const start = result.at(-1)?.end || 0;
  const end = start + (total ? item.total / total * 360 : 0);
  return [...result, { item, index, start, end }];
}, []);

PSEUDOCODE:

calculate sum of displayed categories

  show category amount and percentage

const updateSetting = (key, value) => {
  const next = { ...settings, [key]: value };
  setSettings(next);

  if (key === 'currency') {
    localStorage.setItem('spendwise-currency', value);
  }
  if (key === 'customCategories') {
    localStorage.setItem(
      'spendwise-custom-categories',
      JSON.stringify(value)
    );
  }
  setNotice({ type: 'success', text: 'Settings saved.' });
};

PSEUDOCODE:

receive changed setting
merge it into current settings
update React state
persist setting in local storage
show saved notification
for custom options:
  reject blank, duplicate, or excessive values
  allow rename and delete
