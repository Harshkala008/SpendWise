FILE: routes/authRoutes.js
COMPONENT: Authentication API routes

MAIN CODE EXCERPT:

const authLimit = rateLimit({
  windowMs: 15 * 60 * 1000,
  limit: 10,
  standardHeaders: 'draft-8',
  legacyHeaders: false
});

const credentials = [
  body('password')
    .isLength({ min: 8, max: 128 })
    .withMessage('Password must be 8-128 characters')
];

router.post(
  '/signup',
  authLimit,
  [
    body('email').isEmail().normalizeEmail(),
    ...credentials
  ],
  validate,
  signup
);

router.post('/login', authLimit, [...], validate, login);
router.post('/refresh', refresh);
router.post('/logout', requireAuth, logout);

PSEUDOCODE:

for signup and login:
  apply rate limit
  validate credentials
  if validation succeeds, call controller

for refresh:
  call refresh-token controller

for logout:
  require valid access token
  clear stored refresh-token hash and cookie

FILE: middleware/errorHandler.js
COMPONENT: Central error response

MAIN CODE EXCERPT:

const errorHandler = (error, request, response, next) => {
  let statusCode = error.statusCode || 500;
  let message = error.message || 'Internal server error';
  let errors = error.errors;

  if (error instanceof mongoose.Error.ValidationError) {
    statusCode = 400;
    message = 'Validation failed';
  }

  if (error instanceof mongoose.Error.CastError) {
    statusCode = 400;
    message = `Invalid ${error.path}`;
  }

  response.status(statusCode).json({
    success: false,
    error: { code: error.code || 'INTERNAL_ERROR', message },
    message,
    ...(errors ? { errors } : {})
  });
};

PSEUDOCODE:

receive error from any middleware or controller
choose status code and safe message
convert database validation errors to HTTP 400
convert invalid database ids to HTTP 400
return consistent JSON error shape
include debug stack only outside production when appropriate
