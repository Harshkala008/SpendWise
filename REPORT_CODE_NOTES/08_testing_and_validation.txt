FILE: tests/expense.test.js
COMPONENT: Backend API smoke tests

DESCRIPTION:
The tests use Supertest to call the Express application directly. They check
the health endpoint, standard 404 behavior, and authentication protection for
expense endpoints.

MAIN CODE EXCERPT:

const request = require('supertest');
const app = require('../app');

test('GET /api/health returns a healthy response', async () => {
  const response = await request(app).get('/api/health');

  expect(response.statusCode).toBe(200);
  expect(response.body.success).toBe(true);
  expect(response.body.data.environment).toBeDefined();
});

test('expense routes require authentication', async () => {
  const response = await request(app)
    .post('/api/expenses')
    .send({ title: 'A', amount: 0, category: 'Invalid' });

  expect(response.statusCode).toBe(401);
  expect(response.body.success).toBe(false);
  expect(response.body.error.code).toBe('AUTH_REQUIRED');
});

load Express app without opening a listening port
send request to health endpoint
verify successful status and response structure
send request to an unknown endpoint
verify standard 404 response
send protected expense request without token
verify request is rejected with authentication error
